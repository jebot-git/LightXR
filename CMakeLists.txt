cmake_minimum_required(VERSION 3.24)
project(LightXR VERSION 0.1.0 LANGUAGES C CXX)
include(GNUInstallDirs)
include(CTest)
set(CMAKE_C_STANDARD 99)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
find_package(PkgConfig REQUIRED)
pkg_check_modules(CORE REQUIRED IMPORTED_TARGET gio-unix-2.0 gobject-2.0 xdmcp xcb)
pkg_check_modules(XKL REQUIRED IMPORTED_TARGET libxklavier x11)
pkg_check_modules(XR REQUIRED IMPORTED_TARGET openxr egl gl)
find_package(Qt6 REQUIRED COMPONENTS Widgets Network Test)
find_package(Python3 REQUIRED COMPONENTS Interpreter)
find_path(PAM_INCLUDE_DIR security/pam_appl.h REQUIRED)
find_path(GCRYPT_INCLUDE_DIR gcrypt.h REQUIRED)
find_library(PAM_LIBRARY NAMES pam libpam.so.0 REQUIRED)
find_library(GCRYPT_LIBRARY NAMES gcrypt libgcrypt.so.20 REQUIRED)
configure_file(lightxr/cmake/config.h.in config.h)
include_directories(${CMAKE_CURRENT_BINARY_DIR} ${PAM_INCLUDE_DIR} ${GCRYPT_INCLUDE_DIR} ${GCRYPT_INCLUDE_DIR}/${CMAKE_LIBRARY_ARCHITECTURE})
set(session_dirs "/usr/local/share/wayland-sessions:/usr/share/wayland-sessions:/usr/share/xsessions")
add_compile_definitions(CONFIG_NAMESPACE="lightxr" CONFIG_DIR="/etc/lightxr" SESSIONS_DIR="${session_dirs}" REMOTE_SESSIONS_DIR="${CMAKE_INSTALL_FULL_DATADIR}/lightxr/remote-sessions")
add_library(lightxr-common STATIC common/configuration.c common/dmrc.c common/privileges.c common/user-list.c)
target_include_directories(lightxr-common PUBLIC common)
target_link_libraries(lightxr-common PUBLIC PkgConfig::CORE)
file(GLOB daemon_sources CONFIGURE_DEPENDS src/*.c)
list(REMOVE_ITEM daemon_sources "${CMAKE_CURRENT_SOURCE_DIR}/src/dm-tool.c" "${CMAKE_CURRENT_SOURCE_DIR}/src/lightdm-guest-session.c")
list(REMOVE_ITEM daemon_sources "${CMAKE_CURRENT_SOURCE_DIR}/src/lightdm.c")
add_library(lightxr-daemon-core STATIC ${daemon_sources})
target_include_directories(lightxr-daemon-core PUBLIC src)
add_executable(lightxr src/lightdm.c)
target_compile_definitions(lightxr-daemon-core PUBLIC SBIN_DIR="${CMAKE_INSTALL_FULL_SBINDIR}" USERS_DIR="/var/lib/lightxr/data" LOG_DIR="/var/log/lightxr" RUN_DIR="/run/lightxr/daemon" CACHE_DIR="/var/cache/lightxr")
target_link_libraries(lightxr-daemon-core PUBLIC lightxr-common ${PAM_LIBRARY} ${GCRYPT_LIBRARY})
target_link_libraries(lightxr PRIVATE lightxr-daemon-core)
file(GLOB client_sources CONFIGURE_DEPENDS liblightdm-gobject/*.c)
add_library(lightxr-client STATIC ${client_sources})
target_include_directories(lightxr-client PUBLIC liblightdm-gobject)
target_link_libraries(lightxr-client PUBLIC lightxr-common PkgConfig::XKL)
add_executable(lightxr-greeter lightxr/src/greeter.cpp)
qt_add_resources(lightxr-greeter background PREFIX "/lightxr" BASE lightxr/assets FILES lightxr/assets/openxr-login-background.png)
target_link_libraries(lightxr-greeter PRIVATE lightxr-client Qt6::Widgets Qt6::Network)
add_executable(lightxr-presenter lightxr/src/presenter.cpp lightxr/src/graphics.cpp)
target_link_libraries(lightxr-presenter PRIVATE PkgConfig::XR Qt6::Core)
target_compile_options(lightxr-greeter PRIVATE -Wall -Wextra)
target_compile_options(lightxr-presenter PRIVATE -Wall -Wextra -Wno-missing-field-initializers)
configure_file(lightxr/packaging/lightxr.service.in lightxr.service @ONLY)
configure_file(lightxr/packaging/lightxr-broker.service.in lightxr-broker.service @ONLY)
configure_file(lightxr/packaging/lightxr.desktop.in lightxr.desktop @ONLY)
configure_file(lightxr/packaging/lightdm.conf.in lightdm.conf @ONLY)
install(TARGETS lightxr RUNTIME DESTINATION ${CMAKE_INSTALL_SBINDIR})
install(TARGETS lightxr-greeter lightxr-presenter RUNTIME DESTINATION ${CMAKE_INSTALL_LIBEXECDIR}/lightxr)
install(PROGRAMS lightxr/broker/lightxr-broker.py lightxr/packaging/lightxr-session DESTINATION ${CMAKE_INSTALL_LIBEXECDIR}/lightxr)
install(FILES "${CMAKE_CURRENT_BINARY_DIR}/lightxr.desktop" DESTINATION ${CMAKE_INSTALL_DATADIR}/xgreeters)
install(FILES "${CMAKE_CURRENT_BINARY_DIR}/lightdm.conf" lightxr/packaging/broker.json data/users.conf data/keys.conf DESTINATION /etc/lightxr)
install(FILES data/org.freedesktop.DisplayManager.conf DESTINATION ${CMAKE_INSTALL_DATADIR}/dbus-1/system.d)
install(FILES lightxr/packaging/pam-lightxr DESTINATION /etc/pam.d RENAME lightxr)
install(FILES data/pam/lightdm-greeter DESTINATION /etc/pam.d RENAME lightxr-greeter)
install(FILES "${CMAKE_CURRENT_BINARY_DIR}/lightxr.service" "${CMAKE_CURRENT_BINARY_DIR}/lightxr-broker.service" DESTINATION ${CMAKE_INSTALL_LIBDIR}/systemd/system)
install(FILES lightxr/packaging/lightxr.sysusers DESTINATION ${CMAKE_INSTALL_LIBDIR}/sysusers.d RENAME lightxr.conf)
install(FILES lightxr/packaging/lightxr.tmpfiles DESTINATION ${CMAKE_INSTALL_LIBDIR}/tmpfiles.d RENAME lightxr.conf)
install(FILES lightxr/assets/README.md DESTINATION ${CMAKE_INSTALL_DOCDIR} RENAME BACKGROUND.md)
install(FILES COPYING.GPL3 COPYING.LGPL2 COPYING.LGPL3 lightxr/README.md lightxr/UPSTREAM.json lightxr/COPYING.graphics DESTINATION ${CMAKE_INSTALL_DOCDIR})
if(BUILD_TESTING)
 add_executable(wayland-session-test lightxr/tests/test-wayland-session.c)
 target_link_libraries(wayland-session-test PRIVATE lightxr-daemon-core)
 add_test(NAME wayland-session COMMAND wayland-session-test /usr/share/wayland-sessions/dome.desktop)
 set_tests_properties(wayland-session PROPERTIES SKIP_RETURN_CODE 77)
 add_test(NAME installed-dome-contract COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/lightxr/tests/test_installed_dome.py)
 set_tests_properties(installed-dome-contract PROPERTIES SKIP_RETURN_CODE 77)
 add_test(NAME broker COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/lightxr/tests/test_broker.py)
 add_test(NAME session-wrapper COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/lightxr/tests/test_session.py)
 add_test(NAME frame-channel COMMAND lightxr-presenter --test-channel)
 add_test(NAME geometry COMMAND lightxr-presenter --test-geometry)
 add_executable(greeter-test lightxr/tests/test-greeter.cpp)
 target_link_libraries(greeter-test PRIVATE lightxr-client Qt6::Widgets Qt6::Network Qt6::Test)
 qt_add_resources(greeter-test test_background PREFIX "/lightxr" BASE lightxr/assets FILES lightxr/assets/openxr-login-background.png)
 add_test(NAME greeter-protocol COMMAND greeter-test)
 set_tests_properties(greeter-protocol PROPERTIES ENVIRONMENT "QT_QPA_PLATFORM=offscreen")
 add_test(NAME daemon-defaults COMMAND lightxr --config=${CMAKE_CURRENT_BINARY_DIR}/lightdm.conf --show-config)
endif()
